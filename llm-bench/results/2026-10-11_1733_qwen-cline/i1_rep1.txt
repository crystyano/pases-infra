from dataclasses import dataclass, field
from decimal import Decimal, ROUND_HALF_UP
from typing import Dict, List

@dataclass(frozen=True)
class Item:
    descricao: str
    quantidade: int
    preco_unitario: Decimal

class Orcamento:
    def __init__(self, cliente: str, frete: Decimal = Decimal("0")) -> None:
        cliente = cliente.strip()
        if not cliente:
            raise ValueError("Cliente não pode ser vazio")
        if frete < 0:
            raise ValueError("Frete não pode ser negativo")
        self._cliente = cliente
        self._frete = frete
        self._itens: List[Item] = []
        self._itens_dict: Dict[str, Item] = {}
        self._desconto_percentual = Decimal("0")
        self._desconto_valor = Decimal("0")

    def adicionar_item(self, item: Item) -> None:
        if not item.descricao.strip():
            raise ValueError("Descrição do item não pode ser vazia")
        if not isinstance(item.quantidade, int) or item.quantidade <= 0:
            raise ValueError("Quantidade deve ser um inteiro maior que zero")
        if not isinstance(item.preco_unitario, Decimal) or item.preco_unitario < 0:
            raise ValueError("Preço unitário deve ser um Decimal maior ou igual a zero")
        
        descricao_normalizada = item.descricao.strip().lower()
        if descricao_normalizada in self._itens_dict:
            existing_item = self._itens_dict[descricao_normalizada]
            if item.preco_unitario != existing_item.preco_unitario:
                raise ValueError("Preço unitário diferente para o mesmo item")
            updated_item = Item(
                descricao=existing_item.descricao,
                quantidade=existing_item.quantidade + item.quantidade,
                preco_unitario=existing_item.preco_unitario
            )
            self._itens_dict[descricao_normalizada] = updated_item
            for i, it in enumerate(self._itens):
                if it.descricao.strip().lower() == descricao_normalizada:
                    self._itens[i] = updated_item
                    break
        else:
            self._itens_dict[descricao_normalizada] = item
            self._itens.append(item)

    def subtotal(self) -> Decimal:
        total = sum(
            Decimal(str(item.quantidade)) * item.preco_unitario for item in self._itens
        )
        return total.quantize(Decimal("0.01"), rounding=ROUND_HALF_UP)

    def aplicar_desconto(self, percentual: Decimal, aprovado: bool = False) -> None:
        if not (0 <= percentual <= 100):
            raise ValueError("Percentual deve estar entre 0 e 100")
        if percentual > 15 and not aprovado:
            raise PermissionError("Desconto acima de 15% requer aprovação")
        
        self._desconto_percentual = percentual
        sub = self.subtotal()
        self._desconto_valor = (sub * percentual / 100).quantize(
            Decimal("0.01"), rounding=ROUND_HALF_UP
        )

    def total(self) -> Decimal:
        sub = self.subtotal()
        total = sub - self._desconto_valor + self._frete
        return total.quantize(Decimal("0.01"), rounding=ROUND_HALF_UP)

    def to_dict(self) -> dict:
        itens_saida = []
        for item in self._itens:
            subtotal_item = (Decimal(str(item.quantidade)) * item.preco_unitario).quantize(
                Decimal("0.01"), rounding=ROUND_HALF_UP
            )
            itens_saida.append({
                "descricao": item.descricao,
                "quantidade": item.quantidade,
                "preco_unitario": f"{item.preco_unitario:.2f}",
                "subtotal": f"{subtotal_item:.2f}"
            })
        
        desconto_percentual = f"{self._desconto_percentual:.2f}"
        desconto_valor = f"{self._desconto_valor:.2f}"
        subtotal_str = f"{self.subtotal():.2f}"
        frete_str = f"{self._frete:.2f}"
        total_str = f"{self.total():.2f}"

        return {
            "cliente": self._cliente,
            "itens": itens_saida,
            "subtotal": subtotal_str,
            "desconto_percentual": desconto_percentual,
            "desconto_valor": desconto_valor,
            "frete": frete_str,
            "total": total_str
        }
