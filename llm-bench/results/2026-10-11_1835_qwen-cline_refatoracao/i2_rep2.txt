def calcular_pedido(pedido):
    itens = pedido["itens"]
    cliente = pedido["cliente"]
    cupom = pedido.get("cupom")
    
    subtotal, peso = _calcular_subtotal_e_peso(itens)
    frete = _calcular_frete(peso, cliente)
    desconto = _calcular_desconto(subtotal, cliente, cupom)
    imposto = _calcular_imposto(subtotal, desconto, cliente)
    
    base = subtotal - desconto
    total = round(base + imposto + frete, 2)
    
    return {
        "subtotal": subtotal,
        "desconto": desconto,
        "frete": round(frete, 2),
        "imposto": imposto,
        "total": total,
        "peso_kg": round(peso, 2)
    }


def _calcular_subtotal_e_peso(itens):
    subtotal = 0
    peso = 0
    for item in itens:
        subtotal += item["preco"] * item["qtd"]
        peso += item["peso_kg"] * item["qtd"]
    return round(subtotal, 2), peso


def _calcular_frete(peso, cliente):
    UF_SP = "SP"
    FAIXAS_FRETE = {
        (0, 5): {UF_SP: 20, "outros": 35},
        (5, 20): {UF_SP: 45, "outros": 70},
        (20, float("inf")): {UF_SP: 90, "outros": 140}
    }
    
    frete = None
    for limite_min, limite_max in FAIXAS_FRETE:
        if limite_min < peso <= limite_max:
            frete = FAIXAS_FRETE[(limite_min, limite_max)][UF_SP if cliente["uf"] == UF_SP else "outros"]
            break
    
    if cliente["tipo"] == "VIP":
        frete *= 0.5
    
    if cupom == "FRETEGRATIS":
        if peso > 50:
            frete = 30
        else:
            frete = 0
            
    return frete


def _calcular_desconto(subtotal, cliente, cupom):
    desconto = 0
    
    if cupom == "BEMVINDO10" and subtotal >= 200:
        desconto = subtotal * 0.10
    
    if cliente["tipo"] == "VIP":
        if subtotal >= 1000:
            desconto += (subtotal - desconto) * 0.05
        elif subtotal >= 500:
            desconto += (subtotal - desconto) * 0.02
            
    return round(desconto, 2)


def _calcular_imposto(subtotal, desconto, cliente):
    ALIQUOTAS = {
        "SP": 0.18,
        "RJ": 0.20,
        "MG": 0.18,
        "outros": 0.12
    }
    
    base = subtotal - desconto
    uf = cliente["uf"]
    aliquota = ALIQUOTAS.get(uf, ALIQUOTAS["outros"])
    
    return round(base * aliquota, 2)
