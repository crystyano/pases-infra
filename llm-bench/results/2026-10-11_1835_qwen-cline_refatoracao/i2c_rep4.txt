def resumo_producao(ordens):
    if not ordens:
        return {"linhas": [], "media_aproveitamento": 0, "total_ordens": 0, "revisar": []}

    dados = []
    revisar = []
    total_aprov = 0
    contagem = 0

    for o in ordens:
        aprov = calcular_aproveitamento(o)
        status = determinar_status(aprov)
        
        if o.get("prioridade", 3) == 1 and aprov < 70:
            revisar.append(o["id"])
            
        if status != "INVALIDA":
            total_aprov += aprov
            contagem += 1
            
        dados.append((o.get("prioridade", 3), -aprov, o["id"], status, aprov))

    dados.sort()
    linhas = [f"{d[2]} | {d[3]} | {d[4]}%" for d in dados]
    
    media = round(total_aprov / contagem, 1) if contagem > 0 else 0
    
    return {
        "linhas": linhas,
        "media_aproveitamento": media,
        "total_ordens": len(ordens),
        "revisar": revisar
    }


def calcular_area_chapa(ordem):
    return ordem["chapa_w"] * ordem["chapa_h"]


def calcular_area_pecas(ordem):
    area_total = 0
    for p in ordem["pecas"]:
        area_total += p["w"] * p["h"] * p["qtd"]
    return area_total


def calcular_aproveitamento(ordem):
    area_chapa = calcular_area_chapa(ordem)
    
    if area_chapa <= 0:
        return 0
    
    area_pecas = calcular_area_pecas(ordem)
    aprov = (area_pecas / area_chapa) * 100
    return round(aprov, 1)


def determinar_status(aproveitamento):
    if aproveitamento > 100:
        return "INVALIDA"
    elif aproveitamento >= 85:
        return "OTIMO"
    elif aproveitamento >= 70:
        return "BOM"
    elif aproveitamento >= 50:
        return "REGULAR"
    else:
        return "RUIM"
